// ============================================================================
// ПОЛНЫЙ СПИСОК ВСТРОЕННЫХ КОМПОНЕНТОВ ЯДРА ICED 0.14 (`iced::widget`)
// ============================================================================
// Этот список содержит все официальные виджеты, доступные в модуле `iced::widget`
// версии 0.14, разбитые по категориям для удобного копирования в ваш генератор UI.
// ============================================================================

/// КАТЕГОРИЯ 1: ИНТЕРАКТИВНЫЕ ЭЛЕМЕНТЫ УПРАВЛЕНИЯ И ФОРМЫ
pub enum InteractiveWidgets {
    /// `button` (Кнопка) — Базовый кликабельный элемент. Генерирует событие `on_press`.
    Button,
    
    /// `text_input` (Поле ввода) — Однострочное поле для ввода текста с поддержкой плейсхолдеров,
    /// выделения, копирования/вставки и фокуса. Генерирует событие `on_input`.
    TextInput,
    
    /// `checkbox` (Флажок) — Двухпозиционный чекбокс (галочка) для выбора опций "включено/выключено".
    Checkbox,
    
    /// `radio` (Переключатель) — Круглая радио-кнопка для выбора строго одного варианта из группы.
    Radio,
    
    /// `slider` (Ползунок) — Горизонтальная полоса с бегунком для плавного выбора числовых значений в диапазоне.
    Slider,
    
    /// `toggler` (Переключатель-триггер) — Стилизованный свитч (тумблер) On/Off, альтернатива чекбоксу.
    Toggler,
    
    /// `pick_list` (Выпадающий список) — Дропдаун-меню для выбора одного элемента из текстового списка.
    PickList,
}

/// КАТЕГОРИЯ 2: КОНТЕЙНЕРЫ, СТРУКТУРА И МАКЕТЫ (LAYOUT)
pub enum LayoutContainers {
    /// `column` (Вертикальный стек) — Размещает дочерние элементы друг под другом. 
    /// Управляет вертикальным шагом (`spacing`) и выравниванием по горизонтали.
    Column,
    
    /// `keyed_column` (Индексированный стек) — Оптимизированная версия `column` для динамических 
    /// списков, где каждый элемент привязан к уникальному ключу для быстрой перерисовки.
    KeyedColumn,
    
    /// `row` (Горизонтальный стек) — Размещает дочерние элементы в одну линию слева направо.
    /// Управляет горизонтальным шагом (`spacing`) и выравниванием по вертикали.
    Row,
    
    /// `container` (Универсальный контейнер) — Ограничивает один дочерний элемент. Используется для 
    /// задания внутренних отступов (`padding`), выравнивания, фонового цвета, рамок и теней.
    Container,
    
    /// `scrollable` (Контейнер прокрутки) — Обёртка, добавляющая вертикальную и/или горизонтальную 
    /// прокрутку (скроллбары) для контента, который не помещается на экране.
    Scrollable,
    
    /// `pane_grid` (Сетка панелей) — Адаптивная разметка, разделяющая экран на независимые регионы 
    /// (как в tmux/i3). Поддерживает интерактивное изменение размеров и перетаскивание панелей мышью.
    PaneGrid,
    
    /// `table` (Таблица) — Полноценный виджет для отображения табличных данных с поддержкой 
    /// фиксированных или адаптивных колонок, шапки таблицы и строк.
    Table,
    
    /// `space` (Распорка / Пустое пространство) — Невидимый виджет, занимающий место. Позволяет 
    /// «расталкивать» элементы по краям макета (например, через `Length::Fill`).
    Space,
}

/// КАТЕГОРИЯ 3: ИНФОРМАЦИОННЫЕ И ТЕКСТОВЫЕ ВИДЖЕТЫ
pub enum DataDisplay {
    /// `text` (Текст) — Простой вывод строки текста на экран с возможностью изменения шрифта и размера.
    Text,
    
    /// `rich_text` (Форматированный текст) — Продвинутый вывод текста, где отдельные слова или 
    /// буквы могут иметь разные стили, цвета, шрифты или начертания в рамках одного абзаца.
    RichText,
    
    /// `markdown` (Рендерер разметки) — Парсит и визуализирует стандартный Markdown-текст, 
    /// автоматически преобразуя заголовки, списки и жирный шрифт в UI-элементы.
    Markdown,
    
    /// `progress_bar` (Полоса прогресса) — Визуальный индикатор выполнения задачи (от 0.0 до 100.0%).
    ProgressBar,
    
    /// `qr_code` (QR-код) — Компонент, который берет строку данных и процедурно генерирует 
    /// и рендерит растровый QR-код прямо на экране приложения.
    QrCode,
    
    /// `rule` (Разделитель) — Горизонтальная или вертикальная линия для визуального разграничения блоков интерфейса.
    Rule,
}

/// КАТЕГОРИЯ 4: ГРАФИКА, РИСОВАНИЕ И МУЛЬТИМЕДИА
pub enum MultimediaAndGraphics {
    /// `image` (Изображение) — Загружает и отображает растровые картинки (PNG, JPEG, WebP и т.д.).
    Image,
    
    /// `svg` (Векторная графика) — Загружает и масштабирует векторные файлы SVG без потери качества (иконки).
    Svg,
    
    /// `canvas` (Холст) — Предоставляет 2D-контекст для процедурного рисования (линии, кривые, геометрия).
    /// *Требует активации feature-флага `canvas`*.
    Canvas,
    
    /// `shader` (Шейдер) — Продвинутый виджет для рендеринга кастомных графических шейдеров 
    /// напрямую через графический конвейер WGPU (для эффектов или сложной 3D/2D анимации).
    Shader,
}

/// КАТЕГОРИЯ 5: ФУНКЦИОНАЛЬНЫЕ ВИДЖЕТЫ-ОБЕРТКИ (ПОВЕДЕНИЕ)
pub enum BehavioralWrappers {
    /// `mouse_area` (Область мыши) — Обертка над любым виджетом, позволяющая перехватывать 
    /// детальные события мыши (клик, наведение, уход курсора, правый клик).
    MouseArea,
    
    /// `opaque` (Непрозрачный слой) — Блокирует прокликивание сквозь элемент. Клики мыши не проходят 
    /// к виджетам, расположенным под ним на координатной сетке.
    Opaque,
    
    /// `responsive` (Адаптивная область) — Динамический контейнер, который предоставляет доступ к 
    /// текущим размерам выделенного ему окна в пикселях, позволяя менять макет на лету (мобильный/десктопный вид).
    Responsive,
    
    /// `lazy` (Ленивое обновление) — Кэширует граф дочерних виджетов. Перерисовывает внутренности 
    /// только в том случае, если изменились отслеживаемые зависимости (мощный инструмент оптимизации).
    Lazy,
    
    /// `focusable` (Фокусируемый элемент) — Позволяет сделать любой произвольный кастомный элемент 
    /// частью системной цепочки фокуса (навигации по клавише Tab).
    Focusable,
    
    /// `pin` (Абсолютное позиционирование) — Фиксирует дочерний элемент на жестко заданных 
    /// координатах `X` и `Y` относительно родителя, игнорируя правила автоматического макета.
    Pin,
}

/// КАТЕГОРИЯ 6: ФУНКЦИИ-ХЕЛПЕРЫ ДЛЯ БЫСТРОГО ВЫРАВНИВАНИЯ

    "left",             // Прижимает контент к левому краю
    "right",            // Прижимает контент к правому краю
    "top",              // Прижимает контент к верхнему краю
    "bottom",           // Прижимает контент к нижнему краю
    "horizontal_space", // Создает пустое заполнение по горизонтали
    "vertical_space",   // Создает пустое заполнение по вертикали
    "top_left",         // Позиционирует контент в верхний левый угол
    "top_right",        // Позиционирует контент в верхний правый угол
    "right_center",     // Центрирует вертикально, прижимая к правому краю
];
